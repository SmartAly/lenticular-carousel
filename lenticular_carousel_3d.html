为什么会出现竖条和彩虹高光？如何直接用于项目？

1. 灰色蒙层条的物理本质：微透镜竖纹光栅玻璃

你的观察完全正确！这些灰色蒙层竖条本质上就是一层微型柱面透镜阵列（Reeded / Lenticular Fluted Glass）。
• 静止态（未悬停）： 灰色蒙层依然存在，但处于高透明度的轻微纹理状态（近乎隐形，只保留一层高雅的物理玻璃微反光，如截图1所示）。
• 文字在玻璃下方： 文字本身并无竖线，是位于竖纹玻璃下方的独立内容，其可见度直接受光栅玻璃的开合折射控制。

2. 两侧自然隐匿与高斯柔化漫射彩虹光斑（Foil Diffusion & Edge Falloff）

• 两侧自然隐匿（Edge Falloff）：在物理光栅中，只有观察视角落在镜面反射与布拉格衍射窗口中心区域（偏角较小）时彩虹才最亮。当鼠标游移到卡片两侧边缘时，衍射光线不再折射进人眼，彩虹会自动平滑淡出至完全隐匿（近乎看不见），保留纯粹真实的玻璃质感。
• 高斯柔化漫射（Gaussian Blur）：圆柱微透镜对薄膜干涉彩虹光谱具有强烈的焦外漫射虚化效果，彩虹各色谱之间不存在生硬色块边界，而是通过高斯模糊与多层羽化扩散（36px+ Blur）柔和交融，形成通透柔美的全息极光感。

3. 鼠标左右平移时竖条宽度与文字展示的动态律动

这正是光栅画“翻转透光”最精妙的核心光学效应：
• 鼠标在左侧： 左侧光栅条纹由于观察视角斜切被极端压缩变窄，透镜“闭合”，下方的文字被重度遮蔽压缩，因而几乎看不见（如截图4）。
• 鼠标向右平移： 光线入射角随之推展，右侧蒙层竖条逐渐放宽展开，透光孔径打开，文字得以自然展开并完整清晰呈现（如截图3和截图2）。

单文件即插即用 React 组件 (LenticularCard.tsx) 
import React, { useRef, useState, useEffect } from 'react';

export interface LenticularCardProps {
  image: string;
  tag?: string;
  title?: string;
  strips?: number;
  foil?: number;
  blur?: number;
  ridge?: number;
  refraction?: number;
  tilt?: number;
  lift?: number;
  className?: string;
}

export function LenticularCard({
  image,
  tag = 'REPETITION',
  title = 'Concrete Chorus',
  strips = 56,
  foil = 0.50,
  blur = 36,
  ridge = 0.50,
  refraction = 0.32,
  tilt = 14,
  lift = 40,
  className = ''
}: LenticularCardProps) {
  const cardRef = useRef<HTMLDivElement>(null);
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [hover, setHover] = useState(false);
  const mousePos = useRef({ x: 0.5, y: 0.5 });
  const currentPos = useRef({ x: 0.5, y: 0.5 });
  const hoverAmount = useRef(0);
  const animFrame = useRef<number>();

  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
    if (!cardRef.current) return;
    const rect = cardRef.current.getBoundingClientRect();
    mousePos.current = {
      x: (e.clientX - rect.left) / rect.width,
      y: (e.clientY - rect.top) / rect.height
    };
  };

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let active = true;
    const render = () => {
      currentPos.current.x += (mousePos.current.x - currentPos.current.x) * 0.12;
      currentPos.current.y += (mousePos.current.y - currentPos.current.y) * 0.12;
      const targetHover = hover ? 1.0 : 0.0;
      hoverAmount.current += (targetHover - hoverAmount.current) * 0.14;

      const { x: mx, y: my } = currentPos.current;
      const dx = (mx - 0.5) * 2;
      const hAmount = hoverAmount.current;

      const w = canvas.width;
      const h = canvas.height;
      ctx.clearRect(0, 0, w, h);

      // 1. 竖向高斯柔化彩虹全息光柱 (两侧自然隐匿 + 柔美漫射)
      // 侧边衍射衰减：左右两端自然隐退至看不见
      const edgeFade = Math.pow(Math.sin(Math.max(0, Math.min(1, mx)) * Math.PI), 2.2);
      const beamCenterX = w * (0.10 + mx * 0.80);
      const beamWidth = w * (0.42 + 0.20 * foil);
      const halfSpan = beamWidth * 0.5;
      const tiltRad = (-4 + dx * 10) * (Math.PI / 180);

      const x0 = beamCenterX - halfSpan * Math.cos(tiltRad);
      const y0 = h * 0.5 - halfSpan * Math.sin(tiltRad);
      const x1 = beamCenterX + halfSpan * Math.cos(tiltRad);
      const y1 = h * 0.5 + halfSpan * Math.sin(tiltRad);

      const foilIntensity = foil * hAmount * edgeFade;
      if (foilIntensity > 0.003) {
        ctx.save();
        ctx.globalCompositeOperation = 'screen';
        if ('filter' in ctx) ctx.filter = `blur(${Math.max(10, blur)}px)`;

        const grad = ctx.createLinearGradient(x0, y0, x1, y1);
        grad.addColorStop(0.00, 'rgba(160, 0, 255, 0)');
        grad.addColorStop(0.12, `rgba(130, 0, 255, ${foilIntensity * 0.45})`);
        grad.addColorStop(0.25, `rgba(40, 100, 255, ${foilIntensity * 0.70})`);
        grad.addColorStop(0.38, `rgba(0, 210, 255, ${foilIntensity * 0.85})`);
        grad.addColorStop(0.50, `rgba(0, 255, 130, ${foilIntensity * 0.80})`);
        grad.addColorStop(0.63, `rgba(255, 230, 20, ${foilIntensity * 0.85})`);
        grad.addColorStop(0.77, `rgba(255, 90, 10, ${foilIntensity * 0.70})`);
        grad.addColorStop(0.89, `rgba(255, 10, 130, ${foilIntensity * 0.50})`);
        grad.addColorStop(1.00, 'rgba(180, 0, 220, 0)');

        ctx.fillStyle = grad;
        ctx.fillRect(-60, -60, w + 120, h + 120);
        ctx.restore();
      }
      if ('filter' in ctx) ctx.filter = 'none';

      // 2. 竖纹玻璃微透镜光栅条 (静止时保持半透明微弱质感，悬停时动态展开)
      ctx.save();
      for (let i = 0; i < strips; i++) {
        const u0 = i / strips;
        const u1 = (i + 1) / strips;

        // 鼠标在左侧时左边条纹狭窄紧凑；鼠标右滑时右侧条纹放宽展开
        const warp0 = u0 - (1 - mx) * (1 - u0) * Math.sin(u0 * Math.PI) * 0.16 * hAmount + (mx * 0.5) * Math.sin(u0 * Math.PI) * 0.12 * hAmount;
        const warp1 = u1 - (1 - mx) * (1 - u1) * Math.sin(u1 * Math.PI) * 0.16 * hAmount + (mx * 0.5) * Math.sin(u1 * Math.PI) * 0.12 * hAmount;

        const sx = warp0 * w;
        const sw = Math.max(1.0, (warp1 - warp0) * w);

        // 灰色蒙层微透镜身体
        ctx.fillStyle = `rgba(255, 255, 255, ${0.02 + 0.05 * hAmount})`;
        ctx.fillRect(sx, 0, sw, h);

        // 棱镜边缘微反射
        const distToBeam = Math.abs(sx - beamCenterX) / (w * 0.3);
        const spec = 0.07 + (0.15 + 0.85 * Math.max(0, 1 - distToBeam)) * ridge * hAmount;
        ctx.fillStyle = `rgba(255, 255, 255, ${spec * 0.65})`;
        ctx.fillRect(sx, 0, 1.2, h);

        // 凹槽阴影
        ctx.fillStyle = `rgba(0, 0, 0, ${0.05 + 0.16 * hAmount})`;
        ctx.fillRect(sx + sw - 1.2, 0, 1.2, h);
      }
      ctx.restore();

      if (active) animFrame.current = requestAnimationFrame(render);
    };

    render();
    return () => {
      active = false;
      if (animFrame.current) cancelAnimationFrame(animFrame.current);
    };
  }, [hover, strips, foil, ridge]);

  const dx = (currentPos.current.x - 0.5) * 2;
  const dy = (currentPos.current.y - 0.5) * 2;
  const textOpacity = hover ? Math.max(0, Math.min(1.0, (0.05 + currentPos.current.x * 1.15))) : 0;
  const tracking = (currentPos.current.x < 0.45) ? `${(currentPos.current.x - 0.45) * 4}px` : '0px';

  return (
    <div
      ref={cardRef}
      onMouseEnter={() => setHover(true)}
      onMouseLeave={() => {
        setHover(false);
        mousePos.current = { x: 0.5, y: 0.5 };
      }}
      onMouseMove={handleMouseMove}
      className={`relative w-[260px] h-[360px] rounded-[14px] overflow-hidden cursor-pointer ${className}`}
      style={{
        perspective: '1200px',
        transform: hover
          ? `perspective(1200px) rotateX(${-dy * tilt}deg) rotateY(${dx * tilt}deg) translateZ(${lift}px)`
          : 'perspective(1200px) rotateX(0deg) rotateY(0deg) translateZ(0px)',
        transition: hover ? 'transform 0.08s ease-out' : 'transform 0.4s cubic-bezier(0.2, 0, 0.2, 1)',
        boxShadow: hover ? '0 25px 50px -12px rgba(0, 0, 0, 0.8)' : '0 10px 25px -5px rgba(0, 0, 0, 0.4)'
      }}
    >
      {/* 底层照片 */}
      <img src={image} alt={title} className="absolute inset-0 w-full h-full object-cover select-none" />

      {/* 渐变遮罩 (Scrim) */}
      <div className="absolute inset-0 bg-gradient-to-t from-black/85 via-black/20 to-transparent pointer-events-none" />

      {/* 竖纹玻璃光栅与全息彩虹光束 Canvas 蒙层 */}
      <canvas
        ref={canvasRef}
        width={520}
        height={720}
        className="absolute inset-0 w-full h-full pointer-events-none mix-blend-screen"
      />

      {/* 底部文字 (位于灰色蒙层下方，随光栅展开自然显现) */}
      <div
        className="absolute bottom-5 left-5 right-5 text-white pointer-events-none transition-all duration-150"
        style={{
          opacity: textOpacity,
          letterSpacing: tracking,
          transform: `translateY(${hover ? 0 : 10}px)`
        }}
      >
        <span className="block text-[10px] font-bold tracking-widest text-neutral-300 uppercase mb-1">{tag}</span>
        <h3 className="text-xl font-bold tracking-tight">{title}</h3>
      </div>
    </div>
  );
}
